-- ABYSALL: INSTANT LOADER (MVDN / JSDELIVR)
-- ========================================================

-- Использование jsdelivr CDN полностью обходит кэш raw.githubusercontent.com
local BaseUrl = "https://cdn.jsdelivr.net/gh/herodarun/Abyssal@main/"
local targetPath = "Games/Doors/Main.txt"

local url = BaseUrl .. targetPath .. "?nocache=" .. tostring(math.random(100000, 999999))
local success, result = pcall(function()
    return game:HttpGet(url)
end)

if success and result and #result > 0 then
    if result:sub(1, 20):find("404") or result:find("<!DOCTYPE html>") then
        warn("[ABYSALL] Ошибка: файл не найден -> " .. targetPath)
        return
    end

    local exec, err = loadstring(result)
    if exec then
        print("[ABYSALL] Мгновенный запуск Main.txt на PlaceId: " .. tostring(game.PlaceId))
        task.spawn(exec)
    else
        warn("[ABYSALL] Ошибка синтаксиса в " .. targetPath .. ": " .. tostring(err))
    end
else
    warn("[ABYSALL] Ошибка сети при скачивании: " .. targetPath)
end

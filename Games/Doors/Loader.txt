local BaseUrl = "https://raw.githubusercontent.com/herodarun/Abyssal/main/"

local function LoadDoorsScript(path)
    local success, result = pcall(function()
        return game:HttpGet(BaseUrl .. path)
    end)
    
    if success and result and #result > 0 then
        if result:sub(1, 15):find("404") or result:find("<!DOCTYPE html>") then
            warn("[ABYSALL] Ошибка: файл не найден -> " .. path)
            return
        end

        local exec, err = loadstring(result)
        if exec then
            task.spawn(exec)
        else
            warn("[ABYSALL] Ошибка синтаксиса в " .. path .. ": " .. tostring(err))
        end
    else
        warn("[ABYSALL] Ошибка сети при скачивании: " .. path)
    end
end

if game.PlaceId == 6516141723 then
    LoadDoorsScript("Games/Doors/Main.txt")
else
    LoadDoorsScript("Games/Doors/Lobby.txt")
end

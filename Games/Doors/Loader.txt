local BaseUrl = "https://raw.githubusercontent.com/herodarun/Abyssal/main/"

local function LoadDoorsScript(path)
    local url = BaseUrl .. path .. "?nocache=" .. tostring(math.random(100000, 999999))
    local success, result = pcall(function()
        return game:HttpGet(url)
    end)
    
    if success and result and #result > 0 then
        if result:sub(1, 20):find("404") or result:find("<!DOCTYPE html>") then
            warn("[ABYSALL] Ошибка: файл не найден -> " .. path)
            return
        end

        local exec, err = loadstring(result)
        if exec then
            task.spawn(exec)
        else
            warn("[ABYSALL] Ошибка синтаксиса в " .. path .. ": " .. tostring(err))
        end
    else
        warn("[ABYSALL] Ошибка сети при скачивании: " .. path)
    end
end

-- Список всех известных PlaceId для основной игры DOORS (Hotel, Mines, Backrooms)
local MainGamePlaces = {
    [6516141723] = true, -- Старый Отель
    [6839171723] = true, -- Основная игра (Hotel / Mines / Floors)
    [10901622312] = true, -- The Rooms / Backrooms
}

-- Проверяем, находится ли игрок в основной игре
if MainGamePlaces[game.PlaceId] then
    print("[ABYSALL] Обнаружена основная игра DOORS (PlaceId: " .. tostring(game.PlaceId) .. "). Загрузка Main.txt...")
    LoadDoorsScript("Games/Doors/Main.txt")
else
    print("[ABYSALL] Обнаружено Лобби (PlaceId: " .. tostring(game.PlaceId) .. "). Загрузка Lobby.txt...")
    LoadDoorsScript("Games/Doors/Lobby.txt")
end
